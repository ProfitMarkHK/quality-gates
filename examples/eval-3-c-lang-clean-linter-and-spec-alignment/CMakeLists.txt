cmake_minimum_required(VERSION 3.16)
project(jsonparser C)
set(CMAKE_C_STANDARD 11)
add_library(jsonparser src/parse.c)
target_include_directories(jsonparser PUBLIC include)
enable_testing()

target_link_libraries(test_parse jsonparser)
add_test(NAME test_parse COMMAND test_parse)
